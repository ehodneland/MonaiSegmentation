# Workflow-eksempler for aktiv MonaiSegmentation-struktur

# ==========================================================
# Miljøer
# ==========================================================

# UNet, SSM og MONAI-integrert nnUNet:
conda activate monai

# Ekte nnUNet v2:
conda activate nnunet


# ==========================================================
# Trening: UNet / SSM / SwinUNETR / MONAI-integrert nnUNet
# ==========================================================

conda activate monai

# Alle train_UNet_Monai.py-kjoringer skriver terminal-output og traceback til
# training.log i modellmappen. Loggen overskrives ved ny kjøring av samme
# modellmappe. training_metrics.csv og plots lagres også i modellmappen.

# Trening med én kanal
python3 training/train_UNet_Monai.py \
  --modalities vibe2min \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name UNet

# Trening med tre kanaler
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name UNet

# Trening med to kanaler
python3 training/train_UNet_Monai.py \
  --modalities T2 ADC \
  --reference T2 \
  --datasets bergen \
  --gpu 1 \
  --model_name UNet

# Trening med Bergen og Montpelier
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen mont \
  --gpu 1 \
  --model_name UNet

# Trening av SSM
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name SSM

# Trening av SwinUNETR fra scratch
# Merk: SwinUNETR er arkitekturen. Uten --pretrained starter den med tilfeldige vekter.
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name SwinUNETR \
  --swin_feature_size 48 \
  --batch_size 1 \
  --img_size 192 \
  --lr 1e-4

# Trening av MONAI-integrert nnUNet
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name nnUNet \
  --batch_size 3


# ==========================================================
# Fine-tuning: UNet / SSM / SwinUNETR / MONAI-integrert nnUNet
# ==========================================================

conda activate monai

# Fine-tuning fra lokal UNet-modell
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --model_name UNet \
  --pretrained models/UNet_vibe2min_T2_ADC_v01_bergen_20260319 \
  --freeze_mode decoder_bottleneck \
  --lr 1e-4

# Fine-tuning fra HF UNet-modell
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --model_name UNet \
  --pretrained ehodneland/UNet_vibe2min_T2_ADC_v01_bergen_20260319 \
  --freeze_mode decoder_bottleneck \
  --lr 1e-4

# Fine-tuning fra HF SSM-modell
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --model_name SSM \
  --pretrained ehodneland/SSM_vibe2min_T2_ADC_v01_bergen_20260520 \
  --freeze_mode decoder_bottleneck \
  --lr 1e-4

# Last ned MONAI SwinUNETR BraTS21 MR-checkpoint
# Kilde: MONAI SwinUNETR BraTS21-modelltabell:
# https://github.com/Project-MONAI/research-contributions/tree/main/SwinUNETR/BRATS21
# Direkte lenke til fold 1 fra tabellen:
mkdir -p models/pretrained/swinunetr_brats21_fold1
wget -O /tmp/swinunetr_brats21_fold1.zip \
  https://github.com/Project-MONAI/MONAI-extra-test-data/releases/download/0.8.1/fold1_f48_ep300_4gpu_dice0_9059.zip
unzip /tmp/swinunetr_brats21_fold1.zip -d models/pretrained/swinunetr_brats21_fold1
find models/pretrained/swinunetr_brats21_fold1 -type f

# Fine-tuning fra MONAI SwinUNETR BraTS21 MR-checkpoint
# BraTS21-modellen bruker 4 MR-kanaler og 3 output-klasser; koden laster
# kompatible vekter og hopper over input-/output-lag med ulik shape.
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1 \
  --model_name SwinUNETR \
  --swin_feature_size 48 \
  --pretrained models/pretrained/swinunetr_brats21_fold1 \
  --freeze_mode decoder \
  --batch_size 1 \
  --img_size 192 \
  --lr 1e-5 \
  --epochs 100 \
  --patience 25

# Fine-tuning fra HF MONAI-integrert nnUNet
python3 training/train_UNet_Monai.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --model_name nnUNet \
  --pretrained ehodneland/nnUNet_vibe2min_T2_ADC_v01_bergen_20260514 \
  --freeze_mode decoder_bottleneck \
  --batch_size 3 \
  --lr 1e-4


# ==========================================================
# Trening og fine-tuning: ekte nnUNet v2
# ==========================================================

conda activate nnunet

# Trening av ekte nnUNet v2
python3 training/train_nnUNet_v2.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets bergen \
  --gpu 1

# Fine-tuning av ekte nnUNet v2 fra lokal modell
python3 training/train_nnUNet_v2.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --dataset_id 731 \
  --pretrained models/nnUNetv2_vibe2min_T2_ADC_v01_bergen_Dataset730_EC_20260518

# Fine-tuning av ekte nnUNet v2 fra HF-modell
python3 training/train_nnUNet_v2.py \
  --modalities vibe2min T2 ADC \
  --reference vibe2min \
  --datasets mont \
  --gpu 1 \
  --dataset_id 731 \
  --pretrained ehodneland/nnUNetv2_vibe2min_T2_ADC_v01_bergen_Dataset730_EC_20260518
